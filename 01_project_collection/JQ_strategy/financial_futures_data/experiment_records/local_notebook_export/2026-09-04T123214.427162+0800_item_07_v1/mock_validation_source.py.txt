def get_price(code, **kwargs):
    calls.append((code,kwargs))
    if scenario == 'first_api_failure' or (scenario == 'api_failure' and len(calls)==2):
        raise RuntimeError('synthetic API failure')
    if kwargs['frequency'] == 'daily':
        index = pd.DatetimeIndex(['2024-06-28'])
    else:
        index = pd.date_range('2024-06-28 09:31','2024-06-28 11:30',freq='min').append(pd.date_range('2024-06-28 13:01','2024-06-28 15:00',freq='min'))
    fields=kwargs['fields']
    values={'open':3.0,'high':4.0,'low':2.0,'close':3.0,'volume':10.0,'money':100.0,'open_interest':20.0,'pre_close':np.nan}
    frame=pd.DataFrame({field:[values[field]]*len(index) for field in fields},index=index)
    if kwargs['frequency']=='1m' and code=='IF2409.CCFX':
        if scenario == 'partial_and_empty':
            frame=frame.loc[(frame.index < pd.Timestamp('2024-06-28 13:00')) & (frame.index != pd.Timestamp('2024-06-28 10:00'))]
        elif scenario == 'empty_index':
            frame=pd.DataFrame({field:pd.Series(dtype='float64') for field in fields}, index=pd.Index([],dtype='float64'))
        elif scenario == 'duplicate':
            frame=pd.concat([frame.iloc[:1],frame])
        elif scenario == 'extra_key':
            extra=frame.iloc[:1].copy()
            extra.index=pd.DatetimeIndex(['2024-06-28 09:30'])
            frame=pd.concat([extra,frame])
        elif scenario in ('required_null','infinity','negative','boolean'):
            field='volume' if scenario in ('negative','boolean') else 'open'
            value={'required_null':np.nan,'infinity':np.inf,'negative':-1.0,'boolean':True}[scenario]
            frame[field]=frame[field].astype(object)
            frame.iloc[0,frame.columns.get_loc(field)]=value
        elif scenario=='wrong_columns':
            frame=frame.drop(columns=['open'])
        elif scenario=='source_warning':
            warnings.warn('synthetic possible truncation warning',RuntimeWarning)
        elif scenario=='ohlc_warning':
            frame.iloc[0,frame.columns.get_loc('high')]=1.0
    return frame
